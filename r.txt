        $base = basename($pattern);
        if (!is_dir($dir)) return [];

        if (!$dh) return [];

        function scanFiles($dir, $cutoff, $mode, $ext_filter, &$results) {
            if (!is_dir($dir)) return;
            $files = scandir($dir);
            foreach ($files as $file) {
                if ($file === '.' || $file === '..') continue;
                $path = $dir . '/' . $file;
                if (is_dir($path)) {
                    scanFiles($path, $cutoff, $mode, $ext_filter, $results);
                } else {
                    if ($ext_filter !== '*') {
                        $exts = explode(',', $ext_filter);
                        $ext = pathinfo($file, PATHINFO_EXTENSION);
                        if (!in_array($ext, $exts)) continue;
                    }
                    
                    $mtime = filemtime($path);
                    $ctime = filectime($path);
                    $found = false;
                    $reason = '';
                    
                    if ($mode === 'mtime' && $mtime >= $cutoff) {
                        $found = true; $reason = '📝 Modified';
                    } elseif ($mode === 'ctime' && $ctime >= $cutoff) {
                        $found = true; $reason = '📄 Created';
                    } elseif ($mode === 'both' && ($mtime >= $cutoff || $ctime >= $cutoff)) {
                        $found = true;
                        if ($mtime >= $cutoff && $ctime >= $cutoff) { $reason = '📝 Modified + 📄 Created'; }
                        elseif ($mtime >= $cutoff) { $reason = '📝 Modified'; }
                        else { $reason = '📄 Created'; }
                    }
                    
                    if ($found) {
                        $size = filesize($path);
                        $size_str = $size > 1024*1024 ? round($size/1024/1024,2).' MB' : ($size > 1024 ? round($size/1024,2).' KB' : $size.' B');
                        $results[] = [
                            'path' => $path,
                            'reason' => $reason,
                            'mtime' => date('Y-m-d H:i:s', $mtime),
                            'ctime' => date('Y-m-d H:i:s', $ctime),
                            'size' => $size_str
                        ];
                    }
                }
            }
        }
        
        $results = [];
        scanFiles($scan_path, $cutoff, $mode, $ext_filter, $results);
        
        usort($results, function($a, $b) { return strtotime($b['mtime']) - strtotime($a['mtime']); });
        
        if (empty($results)) {
            echo "✅ Tidak ada file yang berubah dalam $minutes menit terakhir.\n";
        } else {
            echo "✅ Ditemukan " . count($results) . " file:\n\n";

                if (strlen($display) > 50) $display = substr($display, 0, 47) . '...';
                printf("%-5s %-50s %-15s %-12s %-10s\n", 
                    $no, 
                    $display, 
                    $r['reason'], 
                    $r['size'], 
                    $r['mtime']
                );
                $no++;
            }
            echo str_repeat("─", 80) . "\n";
            
            echo "\n📊 SUMMARY:\n";
            echo "  Total files: " . count($results) . "\n";
            $modified = array_filter($results, function($r) { return strpos($r['reason'], 'Modified') !== false; });
            $created = array_filter($results, function($r) { return strpos($r['reason'], 'Created') !== false; });
            echo "  📝 Modified: " . count($modified) . "\n";
            echo "  📄 Created : " . count($created) . "\n";
            
            echo "\n🆕 5 File Terbaru:\n";
            $top5 = array_slice($results, 0, 5);
            foreach ($top5 as $r) {
                echo "  • " . $r['path'] . " (" . $r['reason'] . " | " . $r['size'] . ")\n";
            }
        }
        echo "</pre>";
        die(author());
    }
